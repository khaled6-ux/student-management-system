cmake_minimum_required(VERSION 3.25)
project(student_management_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Crow CONFIG REQUIRED)
find_package(libpqxx CONFIG REQUIRED)
add_executable(student-api backend/src/main.cpp backend/src/student_service.cpp backend/src/student_repository.cpp)
target_include_directories(student-api PRIVATE backend/include)
target_link_libraries(student-api PRIVATE Crow::Crow libpqxx::pqxx)
if(MSVC)
  target_compile_options(student-api PRIVATE /W4 /permissive-)
else()
  target_compile_options(student-api PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
